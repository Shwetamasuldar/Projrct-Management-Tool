/* =============================================================================
   TASKMASTER PRO - COMPLETE PROJECT MANAGEMENT SYSTEM
   5 Sections: Auth → Projects → Kanban → Comments → Notifications
   ============================================================================= */

class TaskMasterPro {
    constructor() {
        this.db = JSON.parse(localStorage.getItem('taskmasterDB')) || {
            users: [],
            projects: [
                {id:1, name:'Website Redesign', desc:'Complete UI/UX overhaul', color:'#ff6b35', owner:1, members:[1,2,3], tasks:[], created:'2026-01-01'},
                {id:2, name:'Mobile App v2.0', desc:'iOS/Android app update', color:'#4ecdc4', owner:1, members:[1,2], tasks:[], created:'2026-01-02'}
            ],
            tasks: [
                {id:1, projectId:1, title:'Design homepage', desc:'Create wireframes and mockups', priority:'high', assignee:2, status:'todo', created:'2026-01-01'},
                {id:2, projectId:1, title:'API Integration', desc:'Connect frontend to backend APIs', priority:'medium', assignee:3, status:'inProgress', created:'2026-01-01'},
                {id:3, projectId:2, title:'Onboarding flow', desc:'Create user onboarding screens', priority:'low', assignee:2, status:'done', created:'2026-01-02'}
            ],
            comments: [],
            notifications: [],
            currentUser: null
        };
        this.currentProject = null;
        this.init();
    }
    
    init() {
        this.saveDB();
        if (!this.db.currentUser) {
            this.showSection('auth');
        } else {
            this.updateUI();
        }
    }
    
    saveDB() {
        localStorage.setItem('taskmasterDB', JSON.stringify(this.db));
    }
    
    // SECTION 1: AUTH SYSTEM
    register(email, password) {
        if (this.db.users.find(u => u.email === email)) {
            return false;
        }
        const user = {
            id: Date.now(),
            email,
            name: email.split('@')[0].replace(/./, c => c.toUpperCase()),
            avatarColor: ['#ff6b35', '#4ecdc4', '#f7931e', '#48bb78'][Math.floor(Math.random()*4)]
        };
        this.db.users.push(user);
        this.db.currentUser = user.id;
        this.saveDB();
        this.updateUI();
        return true;
    }
    
    login(email, password) {
        const user = this.db.users.find(u => u.email === email);
        if (user) {
            this.db.currentUser = user.id;
            this.saveDB();
            this.updateUI();
            return true;
        }
        return false;
    }
    
    logout() {
        this.db.currentUser = null;
        this.saveDB();
        location.reload();
    }
    
    getCurrentUser() {
        return this.db.users.find(u => u.id === this.db.currentUser);
    }
    
    // SECTION 2: PROJECT MANAGEMENT
    createProject(name, description) {
        const project = {
            id: Date.now(),
            name,
            desc: description,
            color: ['#ff6b35', '#4ecdc4', '#f7931e', '#48bb78'][Math.floor(Math.random()*4)],
            owner: this.db.currentUser,
            members: [this.db.currentUser],
            tasks: [],
            created: new Date().toISOString()
        };
        this.db.projects.push(project);
        this.saveDB();
        this.renderProjects();
    }
    
    // SECTION 3: TASK MANAGEMENT
    createTask(projectId, title, description, priority, assignee) {
        const task = {
            id: Date.now(),
            projectId,
            title,
            description,
            priority,
            assignee,
            status: 'todo',
            created: new Date().toISOString()
        };
        this.db.tasks.push(task);
        const project = this.db.projects.find(p => p.id === projectId);
        project.tasks.push(task.id);
        this.saveDB();
        this.renderKanban(projectId);
    }
    
    updateTaskStatus(taskId, status) {
        const task = this.db.tasks.find(t => t.id === taskId);
        if (task) {
            task.status = status;
            this.saveDB();
        }
    }
    
    // SECTION 4: COMMENTS
    addComment(taskId, content) {
        const comment = {
            id: Date.now(),
            taskId,
            userId: this.db.currentUser,
            content,
            created: new Date().toISOString()
        };
        this.db.comments.push(comment);
        this.saveDB();
    }
    
    // SECTION 5: NOTIFICATIONS
    addNotification(message, type = 'info') {
        this.db.notifications.unshift({
            id: Date.now(),
            message,
            type,
            read: false,
            created: new Date().toISOString()
        });
        this.saveDB();
    }
    
    // RENDERING METHODS
    updateUI() {
        const user = this.getCurrentUser();
        document.getElementById('authButtons').classList.add('hidden');
        document.getElementById('navButtons').classList.remove('hidden');
        document.getElementById('userBtn').textContent = ` ${user.name}`;
        this.showSection('projects');
        this.renderProjects();
    }
    
    renderProjects() {
        const user = this.getCurrentUser();
        const projects = this.db.projects.filter(p => p.owner === user.id || p.members.includes(user.id));
        document.getElementById('projectsContainer').innerHTML = projects.map(project => `
            <div class="project-card" onclick="taskMaster.showKanban(${project.id})">
                <div class="project-color" style="background: ${project.color}"></div>
                <h3>${project.name}</h3>
                <p>${project.desc}</p>
                <div class="project-stats">
                    <span> ${project.members.length} members</span>
                    <span> ${project.tasks.length} tasks</span>
                    <span> ${new Date(project.created).toLocaleDateString()}</span>
                </div>
            </div>
        `).join('') || '<div style="grid-column: 1/-1; text-align:center; padding:60px; color:#718096;"><h3>No projects yet</h3><p>Create your first project to get started!</p></div>';
    }
    
    renderKanban(projectId) {
        this.currentProject = projectId;
        const project = this.db.projects.find(p => p.id === projectId);
        const tasks = this.db.tasks.filter(t => t.projectId === projectId);
        
        const statusOrder = ['todo', 'inProgress', 'done'];
        const lists = {};
        statusOrder.forEach(status => {
            lists[status] = tasks.filter(task => task.status === status);
        });
        
        document.getElementById('kanbanLists').innerHTML = statusOrder.map(status => `
            <div class="kanban-list" ondragover="taskMaster.dragOver(event)" ondrop="taskMaster.drop(event, '${status}')">
                <div class="list-header">
                    <div class="list-title">${status.toUpperCase()}</div>
                    <div class="list-badge">${lists[status].length}</div>
                </div>
                ${lists[status].map(task => `
                    <div class="task-card" draggable="true" ondragstart="taskMaster.dragStart(event, ${task.id})">
                        <div class="task-priority priority-${task.priority}"></div>
                        <h4>${task.title}</h4>
                        <p>${task.description}</p>
                        ${task.assignee ? `<div> ${this.db.users.find(u => u.id === task.assignee)?.name}</div>` : ''}
                    </div>
                `).join('')}
            </div>
        `).join('');
        
        this.showSection('kanban');
    }
    
    dragStart(e, taskId) {
        e.dataTransfer.setData('text/plain', taskId);
    }
    
    dragOver(e) {
        e.preventDefault();
        e.currentTarget.classList.add('drag-over');
    }
    
    drop(e, status) {
        e.preventDefault();
        e.currentTarget.classList.remove('drag-over');
        const taskId = parseInt(e.dataTransfer.getData('text/plain'));
        this.updateTaskStatus(taskId, status);
        this.renderKanban(this.currentProject);
    }
    
    showSection(section) {
        document.querySelectorAll('.section').forEach(s => s.classList.add('hidden'));
        document.getElementById(section + 'Section').classList.remove('hidden');
    }
}

const taskMaster = new TaskMasterPro();

// Global event handlers
document.getElementById('authForm').addEventListener('submit', (e) => {
    e.preventDefault();
    const email = document.getElementById('userEmail').value;
    const password = document.getElementById('userPassword').value;
    const isLogin = document.getElementById('authTitle').textContent.includes('Welcome');
    
    if (isLogin ? taskMaster.login(email, password) : taskMaster.register(email, password)) {
        taskMaster.updateUI();
    } else {
        alert('Invalid credentials or user exists');
    }
});

function showLogin() {
    document.getElementById('authTitle').textContent = 'Welcome Back';
    taskMaster.showSection('auth');
}

function showRegister() {
    document.getElementById('authTitle').textContent = 'Join TaskMaster Pro';
    taskMaster.showSection('auth');
}

function showProjectForm() {
    const name = prompt('Project Name:');
    const desc = prompt('Description:');
    if (name) taskMaster.createProject(name, desc);
}

taskMaster.showKanban = function(projectId) {
    taskMaster.renderKanban(projectId);
};
</script>
